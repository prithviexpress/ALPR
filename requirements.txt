# Core detection + OCR
# paddleocr/paddlepaddle are pinned to the last 2.x line deliberately --
# PaddleOCR 3.x changed its constructor API (use_angle_cls/show_log) and
# call pattern in ways that break worker.py's ocr_image(). Do not bump
# these to 3.x without updating that code to match the new API.
ultralytics==8.3.40
paddleocr==2.7.3
paddlepaddle==2.6.2

# Image handling
# opencv-python pinned to EXACTLY 4.6.0.66, not a preference -- it's the
# only version on PyPI that satisfies both ultralytics==8.3.40
# (opencv-python>=4.6.0) and paddleocr==2.7.3 (opencv-python<=4.6.0.66,
# confirmed from its published wheel metadata) at once. Anything else
# (including newer opencv-python releases) makes `pip install -r
# requirements.txt` fail with ResolutionImpossible. Functionally safe:
# every cv2 call this codebase makes (resize/imencode/imread/imwrite/
# absdiff/Laplacian/cvtColor/copyMakeBorder/rectangle/putText) is a
# long-stable API present unchanged well before 4.6.0.
#
# numpy pinned below 2.0: paddlepaddle==2.6.2 predates NumPy 2.0 and
# compatibility with it is unverified.
opencv-python==4.6.0.66
numpy==1.26.4

# MQTT + HTTP snapshot fetch
paho-mqtt==2.1.0
requests==2.32.3

# HTTP webhook trigger source (config "http_trigger.enabled") -- only
# needed if a camera calls this service directly over HTTP instead of/
# alongside the MQTT VCA-event trigger. waitress serves it (a production
# WSGI server) instead of Flask's own dev server.
flask==3.0.3
waitress==3.0.0
